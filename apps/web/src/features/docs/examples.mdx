## Half Adder

Two bits → sum + carry. The simplest arithmetic circuit. Toggle the switches to see how AND computes carry and XOR computes sum.

export const halfAdderCode = `
const HalfAdder = circuit('HalfAdder', {
  inputs:  { a: bit, b: bit },
  outputs: { sum: bit, carry: bit },
  nodes:   { xor1: Xor, and1: And },
  connect: ({ inputs, outputs, nodes: { xor1, and1 } }) => [
    inputs.a.to(xor1.a), inputs.b.to(xor1.b), xor1.out.to(outputs.sum),
    inputs.a.to(and1.a), inputs.b.to(and1.b), and1.out.to(outputs.carry),
  ],
})

const HalfAdderDemo = circuit('HalfAdderDemo', {
  nodes: { sw_a: Switch, sw_b: Switch, dut: HalfAdder, led_sum: Led, led_carry: Led },
  connect: ({ nodes: { sw_a, sw_b, dut, led_sum, led_carry } }) => [
    sw_a.out.to(dut.a), sw_b.out.to(dut.b),
    dut.sum.to(led_sum.in), dut.carry.to(led_carry.in),
  ],
})`

export const halfAdderDisplay = `const HalfAdder = circuit('HalfAdder', {
  inputs:  { a: bit, b: bit },
  outputs: { sum: bit, carry: bit },
  nodes:   { xor1: Xor, and1: And },

    inputs.a.to(and1.a), inputs.b.to(and1.b), and1.out.to(outputs.carry),
  ],
})`

## Full Adder

Three bits (a + b + carry in) → sum + carry out. Reuses HalfAdder as a composite — the simulator expands it to primitives automatically.

export const fullAdderCode = `
const HalfAdder = circuit('HalfAdder', {
  inputs:  { a: bit, b: bit },
  outputs: { sum: bit, carry: bit },
  nodes:   { xor1: Xor, and1: And },
  connect: ({ inputs, outputs, nodes: { xor1, and1 } }) => [
    inputs.a.to(xor1.a), inputs.b.to(xor1.b), xor1.out.to(outputs.sum),
    inputs.a.to(and1.a), inputs.b.to(and1.b), and1.out.to(outputs.carry),
  ],
})

const FullAdder = circuit('FullAdder', {
  inputs:  { a: bit, b: bit, cin: bit },
  outputs: { sum: bit, cout: bit },
  nodes:   { ha1: HalfAdder, ha2: HalfAdder, or1: Or },
  connect: ({ inputs, outputs, nodes: { ha1, ha2, or1 } }) => [
    inputs.a.to(ha1.a), inputs.b.to(ha1.b),
    ha1.sum.to(ha2.a), inputs.cin.to(ha2.b),
    ha2.sum.to(outputs.sum),
    ha1.carry.to(or1.a), ha2.carry.to(or1.b), or1.out.to(outputs.cout),
  ],
})

const FullAdderDemo = circuit('FullAdderDemo', {
  nodes: { sw_a: Switch, sw_b: Switch, sw_cin: Switch, dut: FullAdder, led_sum: Led, led_cout: Led },
  connect: ({ nodes: { sw_a, sw_b, sw_cin, dut, led_sum, led_cout } }) => [
    sw_a.out.to(dut.a), sw_b.out.to(dut.b), sw_cin.out.to(dut.cin),
    dut.sum.to(led_sum.in), dut.cout.to(led_cout.in),
  ],
})`

export const fullAdderDisplay = `const FullAdder = circuit('FullAdder', {

    inputs.a.to(ha1.a), inputs.b.to(ha1.b),
    ha1.sum.to(ha2.a), inputs.cin.to(ha2.b),
    ha2.sum.to(outputs.sum),
    ha1.carry.to(or1.a), ha2.carry.to(or1.b), or1.out.to(outputs.cout),

})`

## Counter

Increments on each clock cycle. A register feeds its output through an adder that adds 1, and the result loops back. Step the clock to watch it count.

export const counterCode = `
const Counter = circuit('Counter', {
  outputs: { count: bus(8) },
  nodes:    { reg: Register(), adder: Adder(), one: Constant({ value: 1 }) },
  connect: ({ outputs, nodes: { reg, adder, one } }) => [
    reg.q.to(adder.a), one.out.to(adder.b),
    adder.sum.to(reg.data), one.out.to(reg.we),
    reg.q.to(outputs.count),
  ],
})

const CounterDemo = circuit('CounterDemo', {
  nodes: { counter: Counter, display: HexDisplay },
  connect: ({ nodes: { counter, display } }) => [
    counter.count.to(display.in),
  ],
})`

export const counterDisplay = `const Counter = circuit('Counter', {

## Fibonacci Generator

Two registers and one adder produce the Fibonacci sequence every clock tick. A DFlipFlop seed trick injects +1 on the first tick only.

export const fibCode = `
const Fibonacci = circuit('Fibonacci', {
  outputs: { fib: bus(8) },
  nodes:    { reg_a: Register(), reg_b: Register(), adder: Adder(), one: Constant({ value: 1 }), init: DFlipFlop() },
  connect: ({ outputs, nodes: { reg_a, reg_b, adder, one, init } }) => [
    one.out.to(init.d),
    init.q_bar.to(adder.carry_in),
    reg_a.q.to(adder.a), reg_b.q.to(adder.b),
    reg_b.q.to(reg_a.data), one.out.to(reg_a.we),
    adder.sum.to(reg_b.data), one.out.to(reg_b.we),
    reg_b.q.to(outputs.fib),
  ],
})

const FibonacciDemo = circuit('FibonacciDemo', {
  nodes: { fib: Fibonacci, display: HexDisplay },
  connect: ({ nodes: { fib, display } }) => [
    fib.fib.to(display.in),
  ],
})`

export const fibDisplay = `const Fibonacci = circuit('Fibonacci', {
  outputs: { fib: bus(8) },

## Design Patterns

**Combinational** — no state, no clock. Outputs computed directly from inputs. Examples: gates, adders, muxes, ALUs.

**Sequential** — has state and clock. State updates on rising clock edge, outputs reflect current state. Examples: counters, registers, shift registers.

**Hierarchical composition** — build complex circuits from simpler ones. FullAdder from HalfAdder, ALU from Adder+Mux, CPU from ALU+RegisterFile+Memory.

**Feedback loops** — connect a node's output back to its input through a register. The register breaks the combinational loop by introducing a one-cycle delay. This is how counters, state machines, and CPUs work.
